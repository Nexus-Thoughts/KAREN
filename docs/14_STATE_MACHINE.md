# 14 — State Machines

## Student presence
UNKNOWN -> INSIDE -> EXIT_CANDIDATE -> EXIT_VERIFIED -> OUTSIDE -> REENTRY_CANDIDATE -> INSIDE

## Class
SCHEDULED -> OPEN -> VERIFICATION -> ATTENDANCE_OPEN -> CLOSED

## Attendance
UNMARKED -> PRESENT/ABSENT/LATE/EXCUSED -> MODIFIED -> FINALIZED

## Notification
CREATED -> QUEUED -> SENT -> READ
Failure: QUEUED -> FAILED -> RETRYING -> SENT/EXHAUSTED

## Report
REQUESTED -> AUTHORIZED -> DATA_FETCHED -> VALIDATED -> RENDERED -> STORED -> DELIVERED
